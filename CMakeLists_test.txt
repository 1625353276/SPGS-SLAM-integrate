# SuperPoint 集成测试程序的 CMakeLists.txt

cmake_minimum_required(VERSION 3.10)
project(test_sp_integration)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# 设置 OpenCV 路径
set(OpenCV_DIR /usr/local/opencv4.7.0/lib/cmake/opencv4)
find_package(OpenCV REQUIRED)

# 设置 CUDA
find_package(CUDA REQUIRED)

# 设置 ONNX Runtime
set(ONNXRUNTIME_ROOT_DIR /usr/local/ort)
include_directories(${ONNXRUNTIME_ROOT_DIR}/include)
link_directories(${ONNXRUNTIME_ROOT_DIR}/lib)

# 设置 Torch
set(Torch_DIR /usr/local/libtorch/share/cmake/Torch)
find_package(Torch REQUIRED)

# 设置 Eigen3
find_package(Eigen3 REQUIRED)

# 设置 Sophus
set(SOPHUS_DIR /usr/local/include/sophus)
include_directories(${SOPHUS_DIR})

# 设置 Pangolin
find_package(Pangolin REQUIRED)

# 设置 Boost
find_package(Boost REQUIRED COMPONENTS serialization)

# 设置 OpenMP
find_package(OpenMP REQUIRED)

# 源文件
add_executable(test_sp_integration test_sp_integration.cpp)

# 包含目录
target_include_directories(test_sp_integration PRIVATE
    ${OpenCV_INCLUDE_DIRS}
    ${CUDA_INCLUDE_DIRS}
    ${EIGEN3_INCLUDE_DIR}
    ${Pangolin_INCLUDE_DIRS}
    ${Boost_INCLUDE_DIRS}
    ${OpenMP_CXX_INCLUDE_DIRS}
    ${CMAKE_CURRENT_SOURCE_DIR}
    ${CMAKE_CURRENT_SOURCE_DIR}/ORB-SLAM3/include
    ${CMAKE_CURRENT_SOURCE_DIR}/ORB-SLAM3/include/Extractors
    ${CMAKE_CURRENT_SOURCE_DIR}/ORB-SLAM3/include/Matchers
    ${CMAKE_CURRENT_SOURCE_DIR}/ORB-SLAM3/Thirdparty/DBoW3/src
    ${CMAKE_CURRENT_SOURCE_DIR}/ORB-SLAM3/Thirdparty/g2o
)

# 链接库
target_link_libraries(test_sp_integration PRIVATE
    ${OpenCV_LIBS}
    ${CUDA_LIBRARIES}
    ${TORCH_LIBRARIES}
    ${Pangolin_LIBRARIES}
    ${Boost_LIBRARIES}
    ${OpenMP_CXX_LIBRARIES}
    ${CMAKE_CURRENT_SOURCE_DIR}/lib/libORB_SLAM3.so
    onnxruntime
    stdc++fs
)

# CUDA 相关设置
set(CUDA_SEPARABLE_COMPILATION ON)
set(CUDA_PROPAGATE_HOST_FLAGS OFF)
